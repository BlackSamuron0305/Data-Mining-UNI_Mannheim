\begin{frame}{Which games? Three designs}
  \small
  \begin{columns}[t]
    \begin{column}{.32\textwidth}
      \hl{Fully random}\\[.3em]
      \begin{tikzpicture}[scale=.32]
        \pgfmathsetseed{7}\foreach \i in {0,...,27} {\pgfmathsetmacro{\h}{0.6+1.6*rnd} \fill[muted!60] (\i*0.32,0) rectangle +(0.24,\h);}
      \end{tikzpicture}\\[.3em]
      every complete game has the same chance\\[.2em]
      \textcolor{bad}{rare groups (e.g.\ classical, 2000+) vary by chance}\\[.2em]
      $\Rightarrow$ yardstick only
    \end{column}
    \begin{column}{.32\textwidth}
      \hl{Stratified random} \textbf{(stratified run)}\\[.3em]
      \begin{tikzpicture}[scale=.32]
        \foreach \i in {0,...,27} {\pgfmathsetmacro{\h}{0.6+1.6*mod(\i*7,11)/11} \fill[accent!70] (\i*0.32,0) rectangle +(0.24,\h);}
      \end{tikzpicture}\\[.3em]
      28 groups (speed $\times$ rating), each \textbf{exactly} its share of the month; random inside\\[.2em]
      \textcolor{good}{only chance inside the groups}
    \end{column}
    \begin{column}{.32\textwidth}
      \hl{Stratified + balanced} \textbf{(balanced run)}\\[.3em]
      \begin{tikzpicture}[scale=.32]
        \foreach \i in {0,...,27} {\pgfmathsetmacro{\h}{0.6+1.6*mod(\i*7,11)/11} \fill[good!70] (\i*0.32,0) rectangle +(0.24,\h);}
        \draw[bad, thick, dashed] (0,1.3) -- (9,1.3);
      \end{tikzpicture}\\[.3em]
      swap games until all \textbf{756 buckets} have the month's blunder rate\\[.2em]
      \textcolor{bad}{chooses games by their label}
    \end{column}
  \end{columns}
  \vspace{.9em}
  \begin{columns}[t]
    \begin{column}{.48\textwidth}
      \textcolor{good}{\textbf{For balancing:}} the models see the true blunder rate in every combination of
      speed, rating and time left
    \end{column}
    \begin{column}{.48\textwidth}
      \textcolor{bad}{\textbf{Against:}} games \emph{without} blunders are calmer games; the 756 rates are
      fixed, everything else may bend
    \end{column}
  \end{columns}
  \vspace{.6em}
  \centering
  \fcolorbox{accent}{accent!6}{Settled by experiment: balance \textbf{training} games only, score A and B on the \textbf{same test games}}
\end{frame}

% ---------------------------------------------------------------------------------------------- 7 methods
\begin{frame}{Methods}
  \small
  \begin{columns}[T]
    \begin{column}{.5\textwidth}
      \footnotesize
      \begin{tabular}{@{}ll@{}}
        \toprule
        Method & Why \\
        \midrule
        Bucket rate (+ 5 simple ones) & \textbf{baselines} \\
        Naive Bayes & lower reference \\
        Logistic regression & calibrated, explains \\
        Linear SVM (+ Platt) & margin, not likelihood \\
        \textbf{Decision tree} & \textbf{symbolic}: rules \\
        Random forest & lowers variance \\
        Gradient boosting, XGBoost & lower bias \\
        \textbf{Neural network} & \textbf{non-symbolic}, smooth \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{.46\textwidth}
      \footnotesize
      \begin{itemize}
        \item Every learner twice: \hl{context} (38 features) and \hl{engine} (+ 15 Stockfish)
        \item Natural class mix, outputs a \textbf{probability}; a cut-off only afterwards
        \item \hl{Tuning}: Optuna (TPE), 10--30 trials per method, each fitted on 1M moves and scored on the
          validation games (log loss); best settings refitted on all
        \item Second round where a best value sat at the edge of its range
        \item One hold-out, not 5-fold: a 5-fold check of best vs.\ second-best showed ties only
        \item Ablation: every feature group left out / alone
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------------------------- 8 evaluation
\begin{frame}{Evaluation setup}
  \small
  \begin{columns}[t]
    \begin{column}{.48\textwidth}
      \hl{Split by game, stratified by group}
      \begin{itemize}
        \item 80\,\% training (10\,\% of it: validation games) / 20\,\% test
        \item Moves of one game never on both sides
        \item Test games used \textbf{once}; both runs share them
      \end{itemize}
      \hl{Why not accuracy?}
      \begin{itemize}
        \item ``Never a blunder'' is right on $\approx$ 95\,\% of moves
      \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
      \hl{Scores}
      \begin{itemize}
        \item Probabilities: log loss, Brier, calibration error
        \item Order: \textbf{PR-AUC}, ROC-AUC
        \item Decision (cut-off from validation): precision, recall, F1, MCC
        \item \textbf{Top 5\,\%} riskiest moves: precision, recall, lift
        \item 95\,\% intervals: 1,000 bootstrap resamples of whole games
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
